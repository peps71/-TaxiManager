import React, { useState } from 'react';
import { 
  LayoutDashboard, 
  Car, 
  Receipt, 
  PieChart, 
  PlusCircle, 
  TrendingUp, 
  TrendingDown, 
  Wallet,
  Landmark
} from 'lucide-react';

export default function App() {
  const [activeTab, setActiveTab] = useState('dashboard');

  // Dati iniziali basati sui CSV forniti dall'utente
  const [annualStats] = useState({
    entrate: 50060.00,
    uscite: 15650.00,
    tasse: 5140.00,
    netto: 29270.00
  });

  const [monthlyData] = useState([
    { mese: 'Gennaio', entrate: 3400, spese: 1100, netto: 1950 },
    { mese: 'Febbraio', entrate: 3450, spese: 1050, netto: 2040 },
    { mese: 'Marzo', entrate: 3900, spese: 1250, netto: 2250 },
    { mese: 'Aprile', entrate: 3750, spese: 1180, netto: 2185 },
  ]);

  const [expenseCategories] = useState([
    { nome: 'Carburante / Diesel', importo: 6200, percentuale: 28.8, deducibilita: '100% (con scheda/fattura)' },
    { nome: 'Manutenzione & Riparazioni', importo: 2400, percentuale: 11.2, deducibilita: '100%' },
    { nome: 'Tasse, INPS & Licenza', importo: 4800, percentuale: 22.3, deducibilita: 'Deducibile in dichiarazione' },
    { nome: 'Rata Prestito / MFD', importo: 3600, percentuale: 16.7, deducibilita: 'Quota interessi deducibile' },
    { nome: 'Assicurazione & Bollo', importo: 1800, percentuale: 8.4, deducibilita: '100%' },
  ]);

  const [dailyRecords, setDailyRecords] = useState([
    { id: 1, data: '2026-08-17', tipo: 'ENTRATA', categoria: 'Corsa CTT -> Stazione', metodo: 'POS', importo: 30.00 },
    { id: 2, data: '2026-08-17', tipo: 'ENTRATA', categoria: 'Corsa Centro -> Ospedale', metodo: 'Contanti', importo: 14.50 },
    { id: 3, data: '2026-08-17', tipo: 'USCITA', categoria: 'Carburante / Diesel', metodo: 'POS', importo: 102.00 },
    { id: 4, data: '2026-08-17', tipo: 'ENTRATA', categoria: 'Corsa Fiera -> Hotel', metodo: 'Fattura', importo: 32.00 },
    { id: 5, data: '2026-08-17', tipo: 'USCITA', categoria: 'Pasti / Lavoro', metodo: 'Contanti', importo: 15.00 },
  ]);

  // Stati per il nuovo inserimento
  const [newRecord, setNewRecord] = useState({ data: '', tipo: 'ENTRATA', categoria: '', metodo: 'POS', importo: '' });
  const [showForm, setShowForm] = useState(false);

  // Formattatore valuta
  const formatCurrency = (value) => {
    return new Intl.NumberFormat('it-IT', { style: 'currency', currency: 'EUR' }).format(value);
  };

  const handleAddRecord = (e) => {
    e.preventDefault();
    if (!newRecord.data || !newRecord.categoria || !newRecord.importo) return;

    const record = {
      id: dailyRecords.length + 1,
      ...newRecord,
      importo: parseFloat(newRecord.importo)
    };

    setDailyRecords([record, ...dailyRecords]);
    setNewRecord({ data: '', tipo: 'ENTRATA', categoria: '', metodo: 'POS', importo: '' });
    setShowForm(false);
  };

  return (
    <div className="min-h-screen bg-gray-50 flex flex-col md:flex-row font-sans text-gray-800">
      
      {/* Sidebar di Navigazione */}
      <aside className="w-full md:w-64 bg-yellow-400 text-gray-900 flex flex-col shadow-lg z-10 md:min-h-screen">
        <div className="p-6 flex items-center space-x-3 bg-yellow-500">
          <Car size={32} className="text-gray-900" />
          <h1 className="text-2xl font-bold tracking-tight">TaxiManager</h1>
        </div>
        <nav className="flex-1 p-4 space-y-2 flex md:flex-col overflow-x-auto md:overflow-visible">
          <button 
            onClick={() => setActiveTab('dashboard')}
            className={`flex items-center space-x-3 w-full p-3 rounded-lg font-medium transition-colors whitespace-nowrap ${activeTab === 'dashboard' ? 'bg-gray-900 text-yellow-400' : 'hover:bg-yellow-500'}`}
          >
            <LayoutDashboard size={20} />
            <span>Dashboard</span>
          </button>
          <button 
            onClick={() => setActiveTab('registro')}
            className={`flex items-center space-x-3 w-full p-3 rounded-lg font-medium transition-colors whitespace-nowrap ${activeTab === 'registro' ? 'bg-gray-900 text-yellow-400' : 'hover:bg-yellow-500'}`}
          >
            <Receipt size={20} />
            <span>Registro Giornaliero</span>
          </button>
          <button 
            onClick={() => setActiveTab('categorie')}
            className={`flex items-center space-x-3 w-full p-3 rounded-lg font-medium transition-colors whitespace-nowrap ${activeTab === 'categorie' ? 'bg-gray-900 text-yellow-400' : 'hover:bg-yellow-500'}`}
          >
            <PieChart size={20} />
            <span>Spese & Fisco</span>
          </button>
        </nav>
      </aside>

      {/* Contenuto Principale */}
      <main className="flex-1 p-6 md:p-8 overflow-y-auto">
        
        {/* --- VISTA DASHBOARD --- */}
        {activeTab === 'dashboard' && (
          <div className="space-y-6 animate-in fade-in duration-500">
            <header className="mb-8">
              <h2 className="text-3xl font-bold text-gray-800">Bilancio Annuale 2026</h2>
              <p className="text-gray-500">Quadro economico completo e andamento mensile.</p>
            </header>

            {/* KPI Cards */}
            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-6">
              <div className="bg-white p-6 rounded-2xl shadow-sm border border-gray-100 flex items-center justify-between">
                <div>
                  <p className="text-sm text-gray-500 font-medium mb-1">Entrate Lorde</p>
                  <p className="text-2xl font-bold text-gray-800">{formatCurrency(annualStats.entrate)}</p>
                </div>
                <div className="w-12 h-12 bg-green-100 rounded-full flex items-center justify-center text-green-600">
                  <TrendingUp size={24} />
                </div>
              </div>
              
              <div className="bg-white p-6 rounded-2xl shadow-sm border border-gray-100 flex items-center justify-between">
                <div>
                  <p className="text-sm text-gray-500 font-medium mb-1">Spese Operative</p>
                  <p className="text-2xl font-bold text-gray-800">{formatCurrency(annualStats.uscite)}</p>
                </div>
                <div className="w-12 h-12 bg-red-100 rounded-full flex items-center justify-center text-red-600">
                  <TrendingDown size={24} />
                </div>
              </div>

              <div className="bg-white p-6 rounded-2xl shadow-sm border border-gray-100 flex items-center justify-between">
                <div>
                  <p className="text-sm text-gray-500 font-medium mb-1">Stima Tasse / INPS</p>
                  <p className="text-2xl font-bold text-gray-800">{formatCurrency(annualStats.tasse)}</p>
                </div>
                <div className="w-12 h-12 bg-orange-100 rounded-full flex items-center justify-center text-orange-600">
                  <Landmark size={24} />
                </div>
              </div>

              <div className="bg-white p-6 rounded-2xl shadow-sm border border-yellow-400 bg-yellow-50 flex items-center justify-between">
                <div>
                  <p className="text-sm text-yellow-800 font-bold mb-1">Guadagno Netto</p>
                  <p className="text-2xl font-black text-gray-900">{formatCurrency(annualStats.netto)}</p>
                </div>
                <div className="w-12 h-12 bg-yellow-400 rounded-full flex items-center justify-center text-gray-900">
                  <Wallet size={24} />
                </div>
              </div>
            </div>

            {/* Grafico Andamento Mensile (Simulato con CSS) */}
            <div className="bg-white p-6 rounded-2xl shadow-sm border border-gray-100 mt-8">
              <h3 className="text-lg font-bold mb-6">Andamento Entrate vs Spese (Primo Trimestre)</h3>
              <div className="space-y-6">
                {monthlyData.map((data, idx) => (
                  <div key={idx} className="flex flex-col sm:flex-row sm:items-center">
                    <div className="w-24 font-medium text-gray-600 mb-2 sm:mb-0">{data.mese}</div>
                    <div className="flex-1 flex flex-col space-y-1">
                      {/* Barra Entrate */}
                      <div className="flex items-center space-x-2">
                        <div className="h-4 bg-green-400 rounded-full" style={{ width: `${(data.entrate / 5000) * 100}%` }}></div>
                        <span className="text-xs text-gray-500 font-medium">{formatCurrency(data.entrate)}</span>
                      </div>
                      {/* Barra Uscite */}
                      <div className="flex items-center space-x-2">
                        <div className="h-4 bg-red-400 rounded-full" style={{ width: `${(data.spese / 5000) * 100}%` }}></div>
                        <span className="text-xs text-gray-500 font-medium">{formatCurrency(data.spese)}</span>
                      </div>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          </div>
        )}

        {/* --- VISTA REGISTRO GIORNALIERO --- */}
        {activeTab === 'registro' && (
          <div className="space-y-6 animate-in fade-in duration-500">
            <header className="flex flex-col sm:flex-row justify-between items-start sm:items-center mb-8 gap-4">
              <div>
                <h2 className="text-3xl font-bold text-gray-800">Registro Giornaliero</h2>
                <p className="text-gray-500">Inserimento corse, rifornimenti e spese operative.</p>
              </div>
              <button 
                onClick={() => setShowForm(!showForm)}
                className="bg-gray-900 hover:bg-gray-800 text-white px-4 py-2 rounded-lg font-medium flex items-center space-x-2 transition-colors"
              >
                <PlusCircle size={20} />
                <span>Nuova Operazione</span>
              </button>
            </header>

            {/* Form di Inserimento */}
            {showForm && (
              <div className="bg-white p-6 rounded-2xl shadow-sm border border-gray-200 mb-8 animate-in slide-in-from-top-4">
                <h3 className="text-lg font-bold mb-4">Aggiungi Movimento</h3>
                <form onSubmit={handleAddRecord} className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-5 gap-4 items-end">
                  <div>
                    <label className="block text-sm font-medium text-gray-700 mb-1">Data</label>
                    <input type="date" required value={newRecord.data} onChange={e => setNewRecord({...newRecord, data: e.target.value})} className="w-full p-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-yellow-400 focus:border-yellow-400 outline-none" />
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-gray-700 mb-1">Tipo</label>
                    <select value={newRecord.tipo} onChange={e => setNewRecord({...newRecord, tipo: e.target.value})} className="w-full p-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-yellow-400 outline-none">
                      <option value="ENTRATA">Entrata (Corsa)</option>
                      <option value="USCITA">Uscita (Spesa)</option>
                    </select>
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-gray-700 mb-1">Descrizione</label>
                    <input type="text" required placeholder="Es. Corsa Stazione" value={newRecord.categoria} onChange={e => setNewRecord({...newRecord, categoria: e.target.value})} className="w-full p-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-yellow-400 outline-none" />
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-gray-700 mb-1">Importo (€)</label>
                    <input type="number" step="0.01" required placeholder="0.00" value={newRecord.importo} onChange={e => setNewRecord({...newRecord, importo: e.target.value})} className="w-full p-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-yellow-400 outline-none" />
                  </div>
                  <div>
                    <button type="submit" className="w-full bg-yellow-400 hover:bg-yellow-500 text-gray-900 font-bold py-2 px-4 rounded-lg transition-colors">
                      Salva
                    </button>
                  </div>
                </form>
              </div>
            )}

            {/* Tabella Registri */}
            <div className="bg-white rounded-2xl shadow-sm border border-gray-100 overflow-hidden">
              <div className="overflow-x-auto">
                <table className="w-full text-left border-collapse">
                  <thead>
                    <tr className="bg-gray-50 border-b border-gray-100 text-gray-500 text-sm">
                      <th className="p-4 font-medium">Data</th>
                      <th className="p-4 font-medium">Tipo</th>
                      <th className="p-4 font-medium">Descrizione</th>
                      <th className="p-4 font-medium">Metodo</th>
                      <th className="p-4 font-medium text-right">Importo</th>
                    </tr>
                  </thead>
                  <tbody>
                    {dailyRecords.map((record) => (
                      <tr key={record.id} className="border-b border-gray-50 hover:bg-gray-50 transition-colors">
                        <td className="p-4 text-sm text-gray-600">{record.data}</td>
                        <td className="p-4">
                          <span className={`inline-flex px-2 py-1 text-xs font-bold rounded-full ${record.tipo === 'ENTRATA' ? 'bg-green-100 text-green-700' : 'bg-red-100 text-red-700'}`}>
                            {record.tipo}
                          </span>
                        </td>
                        <td className="p-4 text-sm font-medium text-gray-800">{record.categoria}</td>
                        <td className="p-4 text-sm text-gray-500">{record.metodo}</td>
                        <td className={`p-4 text-right font-bold ${record.tipo === 'ENTRATA' ? 'text-green-600' : 'text-red-600'}`}>
                          {record.tipo === 'ENTRATA' ? '+' : '-'}{formatCurrency(record.importo)}
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            </div>
          </div>
        )}

        {/* --- VISTA CATEGORIE SPESA --- */}
        {activeTab === 'categorie' && (
          <div className="space-y-6 animate-in fade-in duration-500">
            <header className="mb-8">
              <h2 className="text-3xl font-bold text-gray-800">Dettaglio Categorie Spesa</h2>
              <p className="text-gray-500">Incidenza delle uscite e deducibilità fiscale.</p>
            </header>

            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
              {/* Lista Categorie */}
              <div className="lg:col-span-2 bg-white rounded-2xl shadow-sm border border-gray-100 p-6">
                <h3 className="text-lg font-bold mb-6">Incidenza sul Totale Spese (21.500 €)</h3>
                <div className="space-y-6">
                  {expenseCategories.map((cat, idx) => (
                    <div key={idx}>
                      <div className="flex justify-between items-end mb-2">
                        <div>
                          <p className="font-bold text-gray-800">{cat.nome}</p>
                          <p className="text-xs text-gray-500">{cat.deducibilita}</p>
                        </div>
                        <div className="text-right">
                          <p className="font-bold text-gray-900">{formatCurrency(cat.importo)}</p>
                          <p className="text-sm text-gray-500">{cat.percentuale}%</p>
                        </div>
                      </div>
                      <div className="w-full bg-gray-100 rounded-full h-2">
                        <div className="bg-yellow-400 h-2 rounded-full" style={{ width: `${cat.percentuale}%` }}></div>
                      </div>
                    </div>
                  ))}
                </div>
              </div>

              {/* Box Informativo Fiscale */}
              <div className="bg-gray-900 text-white rounded-2xl p-6 shadow-lg h-fit">
                <div className="flex items-center space-x-3 mb-4">
                  <Landmark className="text-yellow-400" size={28} />
                  <h3 className="text-xl font-bold">Promemoria Fiscale</h3>
                </div>
                <ul className="space-y-4 text-sm text-gray-300">
                  <li className="flex items-start">
                    <div className="w-2 h-2 rounded-full bg-yellow-400 mt-1.5 mr-2 flex-shrink-0"></div>
                    <p>Conserva sempre le fatture elettroniche per garantire la deducibilità del 100% sul <strong className="text-white">Carburante</strong>.</p>
                  </li>
                  <li className="flex items-start">
                    <div className="w-2 h-2 rounded-full bg-yellow-400 mt-1.5 mr-2 flex-shrink-0"></div>
                    <p>Le spese per <strong className="text-white">Pasti e Ristorazione</strong> sono deducibili al 75%, ma nel limite del 2% dei ricavi totali annui.</p>
                  </li>
                  <li className="flex items-start">
                    <div className="w-2 h-2 rounded-full bg-yellow-400 mt-1.5 mr-2 flex-shrink-0"></div>
                    <p>La <strong className="text-white">Manutenzione</strong> è interamente deducibile nel limite del 5% del valore dei beni ammortizzabili.</p>
                  </li>
                </ul>
              </div>
            </div>
          </div>
        )}

      </main>
    </div>
  );
}
